cmake_minimum_required(VERSION 3.15)
project(PID VERSION 1.0.0 LANGUAGES CXX)

# 创建库目标（header-only 库）
add_library(PID INTERFACE)

# 设置编译选项（第19行，但确保上面已经创建了 PID target）
target_compile_options(PID INTERFACE
    $<$<OR:$<CXX_COMPILER_ID:GCC>,$<CXX_COMPILER_ID:Clang>>:
        -Wall -Wextra -Wpedantic
    >
)

# 设置 include 目录
target_include_directories(PID INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)

# 设置 C++ 标准
target_compile_features(PID INTERFACE cxx_std_17)
